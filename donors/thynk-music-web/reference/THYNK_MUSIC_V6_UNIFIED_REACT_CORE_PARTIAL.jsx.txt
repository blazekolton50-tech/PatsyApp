import React, { useState, useEffect, useRef } from 'react';

// --- INLINE SYNTH AND DRUM MACHINE NODE ENGINE ---
class AudioSynthesisEngine {
  constructor() {
    this.ctx = null;
    this.isPlaying = false;
    this.bpm = 120;
    this.timerId = null;
    this.currentStep = 0;
    
    // Configurable parameters via knobs
    this.parameters = {
      reverb: 0.3,
      echo: 0.4,
      delay: 0.32,
      flanger: 0.6,
      cutoff: 0.55,
      resonance: 0.48
    };

    // Tracks setup
    this.tracks = {
      vocals: { volume: 0.8, muted: false, solo: false },
      drums: { volume: 0.7, muted: false, solo: false },
      bass: { volume: 0.8, muted: false, solo: false },
      synth: { volume: 0.6, muted: false, solo: false },
    };
  }

  init() {
    if (this.ctx) return;
    const AudioContextClass = window.AudioContext || window.webkitAudioContext;
    this.ctx = new AudioContextClass();
    
    // Create operational node chain
    this.masterGain = this.ctx.createGain();
    this.filterNode = this.ctx.createBiquadFilter();
    
    this.masterGain.gain.setValueAtTime(0.8, this.ctx.currentTime);
    this.filterNode.type = 'lowpass';
    this.updateFilter();

    this.filterNode.connect(this.masterGain);
    this.masterGain.connect(this.ctx.destination);
  }

  updateFilter() {
    if (!this.ctx) return;
    // Map normalized 0-1 values into real audio spectrum metrics
    this.filterNode.frequency.setValueAtTime(this.parameters.cutoff * 4000 + 200, this.ctx.currentTime);
    this.filterNode.Q.setValueAtTime(this.parameters.resonance * 20, this.ctx.currentTime);
  }

  setParam(key, val) {
    this.parameters[key] = parseFloat(val);
    if (key === 'cutoff' || key === 'resonance') {
      this.updateFilter();
    }
  }

  togglePlayback(onStepCallback) {
    this.init();
    if (this.isPlaying) {
      clearInterval(this.timerId);
      this.isPlaying = false;
    } else {
      this.ctx.resume();
      this.isPlaying = true;
      const intervalMs = (60 / this.bpm / 4) * 1000; // 16th Notes matrix sync loop
      this.timerId = setInterval(() => {
        this.triggerStep();
        onStepCallback(this.currentStep);
        this.currentStep = (this.currentStep + 1) % 16;
      }, intervalMs);
    }
    return this.isPlaying;
  }

  triggerStep() {
    // Generate raw procedural oscillator tones if track channels are unmuted
    if (this.tracks.drums.volume > 0 && !this.tracks.drums.muted) {
      if (this.currentStep % 4 === 0) this.triggerKick();
      if (this.currentStep % 4 === 2) this.triggerSnare();
    }
    if (this.tracks.bass.volume > 0 && !this.tracks.bass.muted) {
      if (this.currentStep % 2 === 0) this.triggerBassTone();
    }
    if (this.tracks.synth.volume > 0 && !this.tracks.synth.muted) {
      this.triggerSynthMelody();
    }
  }

  triggerKick() {
    const osc = this.ctx.createOscillator();
    const gain = this.ctx.createGain();
    osc.connect(gain);
    gain.connect(this.filterNode);

    osc.frequency.setValueAtTime(150, this.ctx.currentTime);
    osc.frequency.exponentialRampToValueAtTime(0.01, this.ctx.currentTime + 0.3);
    gain.gain.setValueAtTime(this.tracks.drums.volume, this.ctx.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.01, this.ctx.currentTime + 0.3);

    osc.start();
    osc.stop(this.ctx.currentTime + 0.3);
  }

  triggerSnare() {
    // Procedural noise burst generator representing the snare snap
    const bufferSize = this.ctx.sampleRate * 0.2;
    const buffer = this.ctx.createBuffer(1, bufferSize, this.ctx.sampleRate);
    const data = buffer.getChannelData(0);
    for (let i = 0; i < bufferSize; i++) {
      data[i] = Math.random() * 2 - 1;
    }

    const noise = this.ctx.createBufferSource();
    noise.buffer = buffer;

    const gain = this.ctx.createGain();
    noise.connect(gain);
    gain.connect(this.filterNode);

    gain.gain.setValueAtTime(this.tracks.drums.volume * 0.4, this.ctx.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.01, this.ctx.currentTime + 0.2);

    noise.start();
    noise.stop(this.ctx.currentTime + 0.2);
  }

  triggerBassTone() {
    const osc = this.ctx.createOscillator();
    const gain = this.ctx.createGain();
    osc.type = 'sawtooth';
    osc.connect(gain);
    gain.connect(this.filterNode);

    const freqMatrix =;
    const targetFreq = freqMatrix[Math.floor(this.currentStep / 4) % freqMatrix.length];

    osc.frequency.setValueAtTime(targetFreq, this.ctx.currentTime);
    gain.gain.setValueAtTime(this.tracks.bass.volume * 0.3, this.ctx.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.01, this.ctx.currentTime + 0.2);

    osc.start();
    osc.stop(this.ctx.currentTime + 0.2);
  }

  triggerSynthMelody() {
    if (Math.random() > 0.6) return;
    const osc = this.ctx.createOscillator();
    const gain = this.ctx.createGain();
    osc.type = 'triangle';
    osc.connect(gain);
    gain.connect(this.filterNode);

    const notes =;
    const targetFreq = notes[this.currentStep % notes.length];

    osc.frequency.setValueAtTime(targetFreq, this.ctx.currentTime);
    gain.gain.setValueAtTime(this.tracks.synth.volume * 0.2, this.ctx.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.01, this.ctx.currentTime + 0.15);

    osc.start();
    osc.stop(this.ctx.currentTime + 0.15);
  }
}
